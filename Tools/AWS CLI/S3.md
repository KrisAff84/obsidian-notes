# S3

### Empty Buckets (script)

```bash
buckets=()

for bucket in "${buckets[@]}"
do

    echo "Emptying bucket $bucket..."

    # List all object versions and delete markers
    versions=$(aws s3api list-object-versions \
        --bucket "$bucket" \
        --prefix "index.html" \
        --query '{Objects: Versions[].{Key:Key,VersionId:VersionId}}' \
        --output json)

    delete_markers=$(aws s3api list-object-versions \
        --bucket "$bucket" \
        --prefix "index.html" \
        --query '{Objects: DeleteMarkers[].{Key:Key,VersionId:VersionId}}' \
        --output json)

    # Combine the results
    combined_json=$(jq -n \
        --argjson versions "$versions" \
        --argjson delete_markers "$delete_markers" \
        '{Objects: ($versions.Objects + $delete_markers.Objects)}')

    # Delete all versions and delete markers
    aws s3api delete-objects \
        --bucket "$bucket" \
        --delete "$combined_json"
    echo "Completed deletion for $bucket."

done
```

### Create Versioned Bucket In Region

```shell
aws s3api create-bucket \
--bucket test-bucket-3878 \
--region us-west-1 \
--create-bucket-configuration LocationConstraint=us-west-1 \
--profile kris84

aws s3api put-bucket-versioning \
--bucket test-bucket-3878 \
--versioning-configuration Status=Enabled \
--profile kris84 \
--region us-west-1
```
